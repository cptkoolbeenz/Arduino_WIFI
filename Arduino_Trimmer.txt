/*
  SD Trim Tester for UNO R4
  - Test-only sketch.
  - Reads one raw CSV file from SD: "value, unix_time"
  - Writes one trimmed CSV file to SD using event-window logic.

  Serial usage:
  1) Open Serial Monitor @ 115200
  2) Enter input filename (8.3 format), e.g. DL260310.TXT
  3) Output filename is auto-generated by replacing leading DL with TR.
*/

#include <SPI.h>
#include <SD.h>
#include <math.h>
#include <limits.h>
#include <LiquidCrystal.h>

#ifndef SD_CS_PIN
#define SD_CS_PIN 10
#endif

const uint32_t CALIBRATION_SECONDS = 600UL;      // Always keep first 10 min
const uint32_t START_GUARD_SECONDS = 3600UL;     // Ignore trigger starts in first hour
const uint32_t PRE_EVENT_SECONDS = 180UL;        // Keep 3 min pre-event
const uint32_t POST_EVENT_SECONDS = 600UL;       // Keep 10 min after event trigger
const long TRIGGER_MIN_DELTA = 2000L;
const float CAL_SPLIT_FRACTION = 0.35f;          // split baseline vs calibration plateaus
const float DEBOUNCE_SECONDS = 0.5f;
const uint32_t PROGRESS_ROWS_ANALYZE = 50000UL;
const uint32_t PROGRESS_ROWS_WRITE = 50000UL;

const int MAX_INTERVALS = 128;

const bool printLCD = true;
const int LCD_RS = 9, LCD_EN = 8, LCD_D4 = 5, LCD_D5 = 4, LCD_D6 = 3, LCD_D7 = 2;
LiquidCrystal lcd(LCD_RS, LCD_EN, LCD_D4, LCD_D5, LCD_D6, LCD_D7);
struct CalibrationThresholds;  // forward declaration for Arduino auto-prototype generation

struct Interval {
  uint32_t start_ts;
  uint32_t end_ts;
};

Interval intervals[MAX_INTERVALS];
int intervalCount = 0;

void setLcdLines(const char *line1, const char *line2) {
  if (!printLCD) return;
  lcd.setCursor(0, 0);
  lcd.print("                ");
  lcd.setCursor(0, 0);
  lcd.print(line1);
  lcd.setCursor(0, 1);
  lcd.print("                ");
  lcd.setCursor(0, 1);
  lcd.print(line2);
}

bool readLine(File &f, char *buf, size_t n) {
  if (!f.available()) return false;
  size_t len = f.readBytesUntil('\n', buf, n - 1);
  buf[len] = '\0';
  while (len > 0 && (buf[len - 1] == '\r' || buf[len - 1] == '\n' || buf[len - 1] == ' ' || buf[len - 1] == '\t')) {
    buf[len - 1] = '\0';
    len--;
  }
  return len > 0;
}

bool parseCsvLine(char *line, long &valueOut, uint32_t &tsOut) {
  char *comma = strchr(line, ',');
  if (!comma) return false;
  *comma = '\0';
  char *left = line;
  char *right = comma + 1;

  while (*left == ' ' || *left == '\t') left++;
  while (*right == ' ' || *right == '\t') right++;

  char *end1 = nullptr;
  char *end2 = nullptr;
  long value = strtol(left, &end1, 10);
  unsigned long ts = strtoul(right, &end2, 10);

  if (end1 == left || end2 == right) return false;
  if (ts == 0UL) return false;

  valueOut = value;
  tsOut = (uint32_t) ts;
  return true;
}

void addInterval(uint32_t startTs, uint32_t endTs) {
  if (endTs <= startTs) return;

  if (intervalCount == 0) {
    intervals[0].start_ts = startTs;
    intervals[0].end_ts = endTs;
    intervalCount = 1;
    return;
  }

  Interval &last = intervals[intervalCount - 1];
  if (startTs <= last.end_ts) {
    if (endTs > last.end_ts) last.end_ts = endTs;
    return;
  }

  if (intervalCount >= MAX_INTERVALS) {
    Serial.println(F("WARN: interval capacity reached; merging into last."));
    if (endTs > last.end_ts) last.end_ts = endTs;
    return;
  }

  intervals[intervalCount].start_ts = startTs;
  intervals[intervalCount].end_ts = endTs;
  intervalCount++;
}

bool inAnyInterval(uint32_t ts, int &idx) {
  while (idx < intervalCount && ts > intervals[idx].end_ts) {
    idx++;
  }
  if (idx >= intervalCount) return false;
  return ts >= intervals[idx].start_ts && ts <= intervals[idx].end_ts;
}

bool readFilenameFromSerial(char *out, size_t outLen) {
  static char lineBuf[32];
  static size_t n = 0;

  while (Serial.available()) {
    char c = (char) Serial.read();

    if (c == '\r' || c == '\n') {
      if (n == 0) continue;
      lineBuf[n] = '\0';
      strncpy(out, lineBuf, outLen - 1);
      out[outLen - 1] = '\0';
      Serial.print(F("RX: "));
      Serial.println(out);
      n = 0;
      return true;
    }

    if (c == 8 || c == 127) {  // backspace/delete
      if (n > 0) n--;
      continue;
    }

    if (c >= 32 && c <= 126) {
      if (n + 1 < sizeof(lineBuf)) {
        lineBuf[n++] = c;
      }
    }
  }
  return false;
}

void makeTrimFilename(const char *inputName, char *outputName, size_t outLen) {
  strncpy(outputName, inputName, outLen - 1);
  outputName[outLen - 1] = '\0';

  if (strlen(outputName) >= 2 && outputName[0] == 'D' && outputName[1] == 'L') {
    outputName[0] = 'T';
    outputName[1] = 'R';
  }
}

struct CalibrationThresholds {
  bool ok;
  uint32_t firstTs;
  long baselineMean;
  long lowCalibrationMean;
  long enterThreshold;
  long exitThreshold;
};

CalibrationThresholds deriveCalibrationThresholds(const char *inputName) {
  CalibrationThresholds r = {false, 0UL, 0L, 0L, 0L, 0L};

  File in = SD.open(inputName, FILE_READ);
  if (!in) return r;

  char line[96];
  long value = 0;
  uint32_t ts = 0;
  bool haveFirst = false;
  long calMin = 0;
  long calMax = 0;
  uint32_t firstTs = 0;

  // Pass 1: calibration min/max.
  while (readLine(in, line, sizeof(line))) {
    if (!parseCsvLine(line, value, ts)) continue;
    if (!haveFirst) {
      haveFirst = true;
      firstTs = ts;
      calMin = value;
      calMax = value;
    }
    if (ts > firstTs + CALIBRATION_SECONDS) break;
    if (value < calMin) calMin = value;
    if (value > calMax) calMax = value;
  }
  in.close();
  if (!haveFirst) return r;

  long split = calMin + (long)((float)(calMax - calMin) * CAL_SPLIT_FRACTION);
  if (split < calMin + TRIGGER_MIN_DELTA) split = calMin + TRIGGER_MIN_DELTA;

  // Pass 2: baseline mean + perturbation segment means.
  in = SD.open(inputName, FILE_READ);
  if (!in) return r;

  double baselineMean = 0.0;
  uint32_t baselineN = 0;

  bool inSeg = false;
  uint32_t segCount = 0;
  uint32_t segIndex = 0;
  long segVals[1200];
  int segValsN = 0;

  long lowMean = LONG_MAX;

  while (readLine(in, line, sizeof(line))) {
    if (!parseCsvLine(line, value, ts)) continue;
    if (ts > firstTs + CALIBRATION_SECONDS) break;

    if (value <= split) {
      baselineN++;
      double delta = (double)value - baselineMean;
      baselineMean += delta / (double)baselineN;
    }

    bool elevated = (value > split);
    if (elevated) {
      if (!inSeg) {
        inSeg = true;
        segCount = 0;
        segValsN = 0;
      }
      if (segValsN < (int)(sizeof(segVals) / sizeof(segVals[0]))) {
        segVals[segValsN++] = value;
      }
      segCount++;
    } else if (inSeg) {
      // Segment ended; compute middle-40% index slice mean.
      if (segValsN >= 20) {
        int s = (int)(0.30f * (float)segValsN);
        int e = (int)(0.70f * (float)segValsN);
        if (e <= s) e = s + 1;
        long long sum = 0;
        int n = 0;
        for (int i = s; i < e && i < segValsN; i++) {
          sum += segVals[i];
          n++;
        }
        if (n > 0) {
          long m = (long)(sum / n);
          if (m < lowMean) lowMean = m;
        }
      }
      inSeg = false;
      segCount = 0;
      segValsN = 0;
    }
    segIndex++;
  }

  // Final trailing segment.
  if (inSeg && segValsN >= 20) {
    int s = (int)(0.30f * (float)segValsN);
    int e = (int)(0.70f * (float)segValsN);
    if (e <= s) e = s + 1;
    long long sum = 0;
    int n = 0;
    for (int i = s; i < e && i < segValsN; i++) {
      sum += segVals[i];
      n++;
    }
    if (n > 0) {
      long m = (long)(sum / n);
      if (m < lowMean) lowMean = m;
    }
  }

  in.close();

  if (baselineN < 10) return r;
  long baseline = (long)baselineMean;
  if (lowMean == LONG_MAX || lowMean <= baseline + TRIGGER_MIN_DELTA) {
    lowMean = baseline + TRIGGER_MIN_DELTA;
  }

  r.ok = true;
  r.firstTs = firstTs;
  r.baselineMean = baseline;
  r.lowCalibrationMean = lowMean;
  r.enterThreshold = lowMean;
  r.exitThreshold = lowMean;
  return r;
}

bool buildIntervals(const char *inputName) {
  intervalCount = 0;
  setLcdLines("Analyze", "Pass start");
  unsigned long tStartMs = millis();

  CalibrationThresholds cal = deriveCalibrationThresholds(inputName);
  if (!cal.ok) {
    Serial.println(F("ERROR: calibration threshold derivation failed."));
    setLcdLines("Analyze FAIL", "Calib failed");
    return false;
  }
  Serial.print(F("Thresholds: baseline="));
  Serial.print(cal.baselineMean);
  Serial.print(F(" lowCal="));
  Serial.print(cal.lowCalibrationMean);
  Serial.print(F(" enter="));
  Serial.print(cal.enterThreshold);
  Serial.print(F(" exit="));
  Serial.println(cal.exitThreshold);

  File in = SD.open(inputName, FILE_READ);
  if (!in) {
    Serial.print(F("ERROR: cannot open input file: "));
    Serial.println(inputName);
    return false;
  }

  char line[96];
  long value = 0;
  uint32_t ts = 0;

  bool haveFirst = false;
  uint32_t firstTs = 0;
  uint32_t lastTs = 0;

  const long enterThreshold = cal.enterThreshold;
  const long exitThreshold = cal.exitThreshold;

  bool eventLatched = false;
  bool captureActive = false;
  uint32_t captureUntil = 0;
  uint32_t currentCaptureStart = 0;
  uint32_t aboveSince = 0;
  uint32_t belowSince = 0;
  uint32_t parsedRows = 0;
  uint32_t nextProgress = PROGRESS_ROWS_ANALYZE;

  while (readLine(in, line, sizeof(line))) {
    if (!parseCsvLine(line, value, ts)) continue;
    parsedRows++;

    if (!haveFirst) {
      haveFirst = true;
      firstTs = ts;
      addInterval(firstTs, firstTs + CALIBRATION_SECONDS);
    }
    lastTs = ts;

    if (ts <= firstTs + CALIBRATION_SECONDS) {
      continue;
    }

    if (ts < firstTs + START_GUARD_SECONDS) {
      continue;
    }

    if (value >= enterThreshold) {
      if (aboveSince == 0) aboveSince = ts;
      belowSince = 0;
    } else if (value <= exitThreshold) {
      if (belowSince == 0) belowSince = ts;
      aboveSince = 0;
    } else {
      aboveSince = 0;
      belowSince = 0;
    }

    if (!eventLatched && aboveSince != 0 && (ts - aboveSince) >= (uint32_t) DEBOUNCE_SECONDS) {
      eventLatched = true;
      uint32_t preStart = (ts > PRE_EVENT_SECONDS) ? (ts - PRE_EVENT_SECONDS) : firstTs;
      if (preStart < firstTs) preStart = firstTs;

      if (!captureActive) {
        captureActive = true;
        currentCaptureStart = preStart;
      } else if (preStart < currentCaptureStart) {
        currentCaptureStart = preStart;
      }
      captureUntil = ts + POST_EVENT_SECONDS;
      aboveSince = 0;

      Serial.print(F("Trigger at ts="));
      Serial.print(ts);
      Serial.print(F(" keep_from="));
      Serial.print(currentCaptureStart);
      Serial.print(F(" keep_until~"));
      Serial.println(captureUntil);
    }

    if (eventLatched && belowSince != 0 && (ts - belowSince) >= (uint32_t) DEBOUNCE_SECONDS) {
      eventLatched = false;
      belowSince = 0;
    }

    if (captureActive && !eventLatched && ts >= captureUntil) {
      addInterval(currentCaptureStart, captureUntil);
      captureActive = false;
    }

    if (parsedRows >= nextProgress) {
      unsigned long elapsed = (millis() - tStartMs) / 1000UL;
      Serial.print(F("Analyze rows="));
      Serial.print(parsedRows);
      Serial.print(F(" intervals="));
      Serial.print(intervalCount);
      Serial.print(F(" elapsed_s="));
      Serial.println(elapsed);
      char l2[17];
      snprintf(l2, sizeof(l2), "r:%lu i:%d", (unsigned long)parsedRows, intervalCount);
      setLcdLines("Analyzing...", l2);
      nextProgress += PROGRESS_ROWS_ANALYZE;
    }
  }

  in.close();

  if (!haveFirst) {
    Serial.println(F("ERROR: no valid rows in input file."));
    return false;
  }

  if (captureActive) {
    uint32_t endTs = (lastTs > captureUntil) ? captureUntil : lastTs;
    addInterval(currentCaptureStart, endTs);
  }

  Serial.print(F("Intervals built: "));
  Serial.println(intervalCount);
  Serial.print(F("Analyze done elapsed_s="));
  Serial.println((millis() - tStartMs) / 1000UL);
  char l2[17];
  snprintf(l2, sizeof(l2), "I:%d", intervalCount);
  setLcdLines("Analyze done", l2);
  return true;
}

bool writeTrimmedFile(const char *inputName, const char *outputName) {
  setLcdLines("Writing trim", "Open files...");
  unsigned long tStartMs = millis();
  File in = SD.open(inputName, FILE_READ);
  if (!in) {
    Serial.print(F("ERROR: cannot reopen input: "));
    Serial.println(inputName);
    setLcdLines("Write FAIL", "No input file");
    return false;
  }

  SD.remove(outputName);
  File out = SD.open(outputName, FILE_WRITE);
  if (!out) {
    in.close();
    Serial.print(F("ERROR: cannot open output: "));
    Serial.println(outputName);
    setLcdLines("Write FAIL", "No output file");
    return false;
  }

  char line[96];
  long value = 0;
  uint32_t ts = 0;
  int idx = 0;
  uint32_t kept = 0;
  uint32_t total = 0;
  uint32_t nextProgress = PROGRESS_ROWS_WRITE;

  while (readLine(in, line, sizeof(line))) {
    char parse[96];
    strncpy(parse, line, sizeof(parse) - 1);
    parse[sizeof(parse) - 1] = '\0';
    if (!parseCsvLine(parse, value, ts)) continue;

    total++;
    if (inAnyInterval(ts, idx)) {
      out.println(line);
      kept++;
    }

    if (total >= nextProgress) {
      unsigned long elapsed = (millis() - tStartMs) / 1000UL;
      Serial.print(F("Write rows="));
      Serial.print(total);
      Serial.print(F(" kept="));
      Serial.print(kept);
      Serial.print(F(" elapsed_s="));
      Serial.println(elapsed);
      char l1[17];
      char l2[17];
      snprintf(l1, sizeof(l1), "Writing %lu", (unsigned long)total);
      snprintf(l2, sizeof(l2), "Kept %lu", (unsigned long)kept);
      setLcdLines(l1, l2);
      nextProgress += PROGRESS_ROWS_WRITE;
    }
  }

  out.close();
  in.close();

  Serial.print(F("Rows total="));
  Serial.print(total);
  Serial.print(F(" kept="));
  Serial.println(kept);
  Serial.print(F("Write done elapsed_s="));
  Serial.println((millis() - tStartMs) / 1000UL);

  File outInfo = SD.open(outputName, FILE_READ);
  if (outInfo) {
    Serial.print(F("Output size bytes="));
    Serial.println((unsigned long) outInfo.size());
    outInfo.close();
  }

  setLcdLines("Trim done", outputName);

  return true;
}

void listRootFiles() {
  File root = SD.open("/");
  if (!root || !root.isDirectory()) {
    Serial.println(F("Cannot open SD root."));
    return;
  }
  Serial.println(F("Files on SD:"));
  while (true) {
    File entry = root.openNextFile();
    if (!entry) break;
    if (!entry.isDirectory()) {
      Serial.print(F("- "));
      Serial.print(entry.name());
      Serial.print(F(" ("));
      Serial.print((unsigned long) entry.size());
      Serial.println(F(" bytes)"));
    }
    entry.close();
  }
  root.close();
}

void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println();
  Serial.println(F("SD Trim Tester start"));
  if (printLCD) {
    lcd.begin(16, 2);
    setLcdLines("Trim Tester", "Init SD...");
  }

  if (!SD.begin(SD_CS_PIN)) {
    Serial.println(F("SD init failed."));
    setLcdLines("SD init FAIL", "Check card");
    while (1) delay(1000);
  }

  listRootFiles();
  Serial.println(F("Enter input filename (8.3):"));
  setLcdLines("Ready", "Enter file name");
}

void loop() {
  static bool waitingForInput = true;
  static char inputName[24] = {0};
  static char outputName[24] = {0};

  if (waitingForInput) {
    if (readFilenameFromSerial(inputName, sizeof(inputName))) {
      waitingForInput = false;
      Serial.print(F("Input set: "));
      Serial.println(inputName);
      makeTrimFilename(inputName, outputName, sizeof(outputName));
      Serial.print(F("Output set: "));
      Serial.println(outputName);
      setLcdLines("Trim start", outputName);

      Serial.println(F("Building keep intervals..."));
      bool ok1 = buildIntervals(inputName);
      if (!ok1) {
        Serial.println(F("Trim failed during interval build."));
        setLcdLines("Trim FAIL", "Analyze step");
      } else {
        Serial.println(F("Writing trimmed file..."));
        bool ok2 = writeTrimmedFile(inputName, outputName);
        if (ok2) {
          Serial.println(F("Trim complete."));
          setLcdLines("Trim complete", outputName);
        } else {
          Serial.println(F("Trim failed during write."));
          setLcdLines("Trim FAIL", "Write step");
        }
      }

      Serial.println();
      listRootFiles();
      Serial.println(F("Enter input filename (8.3):"));
      setLcdLines("Ready", "Enter file name");
      waitingForInput = true;
    } 
    return;
  }
}
